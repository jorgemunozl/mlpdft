\begin{frame}{Goal}
  \begin{itemize}
    \item Summarize \textbf{MACE}: many-body equivariant message-passing potentials (Batatia et al., NeurIPS 2022).
    \item Document two FitSNAP styles: \textbf{linear ridge} on SNAP (\texttt{LiBF4-minimal.in}) and \textbf{PyTorch MLP} on SNAP (\texttt{model/}); compare to \textbf{MACE-MP} finetune on LiF64.
    \item Position a \textbf{fair comparison} and \textbf{MACE-MP finetuning} on the in-repo \textbf{LiF64 NEWJSON} set (\texttt{examples/LiF64\_kjpaw\_v2/NEWJSON/}).
  \end{itemize}
\end{frame}

\begin{frame}{MLIPs: problem statement}
  \begin{itemize}
    \item Need \textbf{DFT-quality} energies, forces, and (for solids) stress at \textbf{MD cost}.
    \item Classical snap fits: \textbf{fixed local basis} + linear or small nonlinear head.
    \item Graph models: learn features on the \textbf{local neighborhood graph} in 3D with \textbf{rotation/reflection equivariance} ($O(3)$ or $E(3)$).
  \end{itemize}
\end{frame}

\begin{frame}{MACE (idea)}
  \begin{itemize}
    \item \textbf{Equivariant MPNN} on atoms; internal features carry angular momentum channels ($L=0$ scalar, $L>0$ tensors).
    \item Key idea: \textbf{explicit many-body messages} via coupled tensor products of neighbor-edge features (correlation order $\nu$), not only stacking many 2-body layers.
    \item Often \textbf{two message-passing layers} suffice when $\nu$ is high enough --- decouples many-body expressivity from deep receptive-field growth.
    \item Readout uses \textbf{invariant} ($L{=}0$) channels so site energies are scalars; sum gives extensive total energy.
  \end{itemize}
\end{frame}

\begin{frame}{MACE (data flow sketch)}
  \[
    \text{atoms } (\mathbf r_i, Z_i)
    \;\Rightarrow\;
    \text{2-body equivariant } A_i
    \;\Rightarrow\;
    \text{tensor products } B_i^{(\nu)}
    \;\Rightarrow\;
    \text{message } m_i
    \;\Rightarrow\;
    \text{update } h_i
    \;\Rightarrow\;
    E_i
  \]
  \vfill
  \begin{center}
    \small
    Neighbor sum retains permutational invariance; Clebsch--Gordan structure enforces $O(3)$ equivariance.
  \end{center}
\end{frame}

\begin{frame}{FitSNAP baseline in \texttt{mlpdft/model/} (PyTorch)}
  \begin{itemize}
    \item \textbf{Solver:} \texttt{PYTORCH} --- \textbf{nonlinear} SNAP fit (\texttt{nonlinear = 1} in the corresponding \texttt{.in}).
    \item \textbf{Descriptors:} SNAP bispectrum (\texttt{LAMMPSSNAP}): \texttt{twojmax = 8 8}, \texttt{chemflag = 1}, \texttt{bikflag / dgradflag} on.
    \item \textbf{Cutoff knobs:} \texttt{rcutfac}, \texttt{rfac0}, \texttt{rmin0}, \texttt{radelem = 0.5 0.5} (LAMMPS-SNAP convention).
    \item \textbf{Head:} PyTorch MLP \texttt{num\_desc $\to$ 256 $\to$ 128 $\to$ 64 $\to$ 64 $\to$ 1}, \texttt{multi\_element\_option = 2} (per-element nets).
    \item \textbf{Training:} learning rate \texttt{5e-6}, \texttt{70} epochs, batch \texttt{50}; weights \texttt{LiF\_Pytorch.pt}.
    \item \textbf{Data (original deck):} \texttt{dataPath = ../LiF\_kjpaw\_chem/JSON} (multi-group; not all in-repo).
    \item \textbf{ML-IAP exports (LiF64):} \texttt{LiF64\_NEWJSON\_pot.*}.
  \end{itemize}
\end{frame}

\begin{frame}{FitSNAP linear baseline (\texttt{configs/fitsnap/LiBF4-minimal.in})}
  \begin{itemize}
    \item \textbf{Solver:} \texttt{RIDGE} --- \textbf{linear} regression, not a deep net (\texttt{nonlinear = 0}).
    \item \textbf{Pipeline:} JSON $\to$ LAMMPS-SNAP bispectrum $\to$ fixed descriptor vector per atom $\to$ ridge fit to DFT energy (forces/stress via descriptor derivatives as in FitSNAP).
    \item \textbf{Species / SNAP:} \texttt{numTypes = 3}, \texttt{type = Li B F}, \texttt{twojmax = 8 8 8}, \texttt{chemflag = 1}, \texttt{bzeroflag = 1}, \texttt{bikflag} / \texttt{dgradflag} on; radial knobs \texttt{rcutfac}, \texttt{rfac0}, \texttt{rmin0}, \texttt{radelem = 0.5} each.
    \item \textbf{Readout:} single \textbf{linear} map from SNAP space (L2-regularized), unlike the PyTorch MLP in \texttt{model/LiF-example.in}.
  \end{itemize}
\end{frame}

\begin{frame}{MACE-MP-0a \texttt{small} (cached checkpoint)}
  \small
  \begin{itemize}
    \item \textbf{Release file (typical cache path):} \texttt{\textasciitilde/.cache/mace/20231210mace128L0\_energy\_epoch249model} (naming: 128 channels, $L{=}0$ / invariant hidden irreps only).
    \item \textbf{Training data:} Materials Project \textbf{MPTrj} trajectories; \textbf{89} elements; VASP PBE+U --- not the same distribution as our QE LiF JSON (architecture comparison only).
    \item \textbf{Parameterization:} \texttt{ScaleShiftMACE} --- atomic reference energies plus learnable scale/shift; then equivariant message passing.
    \item \textbf{Hyperparameters} (from \texttt{extract\_config\_mace\_model} on this checkpoint; re-run \texttt{scripts/print\_mace\_model\_arch.py} after downloads):
      \texttt{r\_max}${}=6.0$~\AA, \texttt{num\_interactions}${}=2$, \textbf{correlation}${}=3$, hidden irreps \texttt{128x0e}, readout MLP \texttt{16x0e}, \texttt{max\_ell}${}=3$, \texttt{num\_bessel}${}=10$, radial MLP $[64,64,64]$, \texttt{avg\_num\_neighbors}${}\approx 62$.
  \end{itemize}
\end{frame}
