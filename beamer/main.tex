\section{Dataset}

\begin{frame}{What dataset we have}
  \begin{itemize}
    \item Large \textbf{DFT-labeled} set for electrolyte / solid-electrolyte style chemistry: mostly \textbf{Li--F}, \textbf{LiBF$_4$}, and \textbf{B--Li} compositions (bulk BLi, BLi$_3$, many \textbf{interface} and \textbf{NPT} variants), plus \textbf{BCC-54-atom} lithium supercells, \textbf{isolated} and \textbf{relaxed} cells, and ionic-liquid-related groups such as \textbf{EMIM-TFSI-BF$_4$} and \textbf{EMIMLiTFSI} anode setups.
    \item \textbf{On the order of 50--60 distinct configuration groups} (folder names), including multiple versions (\texttt{v1}, \texttt{v2}, \texttt{v3}, \texttt{v4}), \texttt{final} and \texttt{kjpaw} tags, LiF 64-atom and interface series (\texttt{LiF64\_…}, \texttt{LiFinterface\_…}), Li-with-F and Li-with-BF clusters, and a catch-all \texttt{special} group.
bts are used day-to-day; the full collection lives in shared project storage under names like \textbf{LiF\_kjpaw}.
    \item Each group holds \textbf{structures with total energies, atomic forces, and virial/stress where available}, suitable for training interatomic potentials (same information can be delivered as extended XYZ or JSON for FitSNAP).
  \end{itemize}
\end{frame}

\section{FitSNAP model}
\begin{frame}{The FitSNAP model we have}
  \begin{itemize}
    \item Training data are \textbf{split by configuration group} (train/test fractions per group); labels are energies, forces, and optionally stress.
    \item \textbf{FitSNAP stack} in words:
      \begin{itemize}
        \item \textbf{Scraper} --- reads quantum-chemistry frames (e.g.\ JSON) into structures.
        \item \textbf{Calculator} --- builds \textbf{SNAP bispectrum} descriptors per atom in LAMMPS (radial cutoff, \texttt{twojmax}, chemical channels, etc.).
        \item \textbf{Solver} --- either \textbf{ridge regression} (linear model on descriptors) or \textbf{PyTorch} (nonlinear MLP on descriptor vectors), depending on the fit setup.
      \end{itemize}
  \end{itemize}
\end{frame}

\section{MACE model}

\begin{frame}{What the MACE model is}
  \begin{enumerate}
    \item \textbf{Variants} --- \textbf{foundation} models (e.g.\ MACE-MP trained on Materials Project trajectories) vs.\ \textbf{finetuned} or trained-from-scratch potentials on your own DFT data.
    \item \textbf{Architecture} --- \textbf{equivariant message passing} on the local neighbor graph, explicit \textbf{many-body correlation} (order $\nu$), radial cutoff \textbf{\texttt{r\_max}}, and typically \textbf{scale--shift} atomic energy handling (\textbf{ScaleShiftMACE}) for materials.
  \end{enumerate}
\end{frame}

\section{Zero-shot evaluation}

\begin{frame}{Zero-shot evaluation of MACE}
  \begin{itemize}
    \item Run a \textbf{pretrained foundation} MACE (e.g.\ the smallest MACE-MP checkpoint) on a \textbf{held-out subset} of the same structures used for FitSNAP testing.
    \item Report \textbf{energy and force errors} in consistent units (eV, eV/\AA) and, if needed, stress, so results are comparable to the SNAP-based baseline on \textbf{identical frames}.
  \end{itemize}
\end{frame}

\section{Finetuning}

\begin{frame}{Challenges for finetuning MACE}
  \begin{itemize}
    \item \textbf{Many-body vs.\ cutoff} --- align MACE interaction cutoff with the SNAP radial setup where comparisons should be fair; correlation order in MACE is not the same object as SNAP descriptor depth.
    \item \textbf{Small data} --- a few dozen to a few hundred DFT frames vs.\ millions in foundation pretraining: risk of overfitting; choose epochs, batch size, and regularization accordingly.
    \item \textbf{Energy reference} --- map DFT total energies to MACE conventions (atomic reference energies, optional \texttt{E0} corrections when starting from a foundation model).
  \end{itemize}
\end{frame}
